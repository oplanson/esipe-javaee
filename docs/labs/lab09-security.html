Hint 2 — the changeIndice 2 — la modification 
@WebFilter(urlPatterns = "/*")
public class SecurityHeadersServletFilter implements Filter {
    @Override
    public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain)
            throws IOException, ServletException {
        HttpServletResponse r = (HttpServletResponse) res;
        r.setHeader("X-Frame-Options", "DENY");
        r.setHeader("X-Content-Type-Options", "nosniff");
        r.setHeader("Content-Security-Policy", "default-src 'self'; frame-ancestors 'none'");
        r.setHeader("Referrer-Policy", "strict-origin-when-cross-origin");
        r.setHeader("Permissions-Policy", "geolocation=(), microphone=(), camera=()");
        if ("https".equals(req.getScheme())) {
            r.setHeader("Strict-Transport-Security", "max-age=31536000; includeSubDomains");
        }
        chain.doFilter(req, res);
    }
}
