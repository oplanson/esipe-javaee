@ApplicationScoped — Singleton service

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.util.List;
import java.util.concurrent.atomic.AtomicLong;

@ApplicationScoped
public class ClientService {

    private final ClientRepository clientRepository;
    private final AtomicLong counter = new AtomicLong(0);

    // Constructor injection — preferred approach
    @Inject
    public ClientService(ClientRepository clientRepository) {
        this.clientRepository = clientRepository;
    }

    public List<Client> findAll() {
        counter.incrementAndGet();
        return clientRepository.findAll();
    }

    public Client create(Client client) {
        counter.incrementAndGet();
        return clientRepository.save(client);
    }

    public long getCallCount() {
        return counter.get();
    }

    @PostConstruct
    void init() {
        System.out.println("ClientService initialized — singleton scope");
    }

    @PreDestroy
    void destroy() {
        System.out.println("ClientService destroyed after " + counter.get() + " calls");
    }
}

CDI Qualifiers

import jakarta.inject.Qualifier;
import java.lang.annotation.*;

// 1. Qualifier definition
@Qualifier
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.METHOD, ElementType.FIELD,
         ElementType.PARAMETER, ElementType.TYPE})
public @interface PremiumClient {}


// 2. Qualified implementation
@PremiumClient
@ApplicationScoped
public class PremiumClientServiceImpl implements ClientNotificationService {

    @Override
    public void notify(Client client, String message) {
        // Premium channel: email + SMS + push notification
        sendEmail(client, message);
        sendSms(client, message);
        sendPushNotification(client, message);
    }
}


// 3. Standard implementation (no qualifier = @Default)
@ApplicationScoped
public class StandardClientServiceImpl implements ClientNotificationService {

    @Override
    public void notify(Client client, String message) {
        // Standard channel: email only
        sendEmail(client, message);
    }
}


// 4. Injection point with qualifier
@ApplicationScoped
public class AccountService {

    @Inject @PremiumClient
    private ClientNotificationService premiumNotifier;

    @Inject
    private ClientNotificationService standardNotifier;
}

Producer Method

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.inject.Disposes;
import jakarta.enterprise.inject.Produces;
import jakarta.persistence.EntityManager;
import jakarta.persistence.EntityManagerFactory;
import jakarta.persistence.Persistence;

@ApplicationScoped
public class EntityManagerProducer {

    private EntityManagerFactory emf;

    @PostConstruct
    void init() {
        emf = Persistence.createEntityManagerFactory("bankPU");
    }

    // Producer: creates an EntityManager injectable via @Inject
    @Produces
    @ApplicationScoped
    public EntityManager produceEntityManager() {
        return emf.createEntityManager();
    }

    // Disposer: closes the EntityManager when it is destroyed
    public void closeEntityManager(@Disposes EntityManager em) {
        if (em.isOpen()) {
            em.close();
        }
    }

    @PreDestroy
    void shutdown() {
        if (emf != null && emf.isOpen()) {
            emf.close();
        }
    }
}

Logging Interceptor

java 
import jakarta.interceptor.InterceptorBinding;
import java.lang.annotation.*;

// 1. Binding annotation
@InterceptorBinding
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.METHOD, ElementType.TYPE})
public @interface Logged {}


// 2. Interceptor implementation
import jakarta.annotation.Priority;
import jakarta.interceptor.AroundInvoke;
import jakarta.interceptor.Interceptor;
import jakarta.interceptor.InvocationContext;
import java.util.logging.Logger;

@Logged
@Interceptor
@Priority(Interceptor.Priority.APPLICATION)
public class LoggingInterceptor {

    private static final Logger logger =
        Logger.getLogger(LoggingInterceptor.class.getName());

    @AroundInvoke
    public Object logMethod(InvocationContext context) throws Exception {
        String className  = context.getTarget().getClass().getSimpleName();
        String methodName = context.getMethod().getName();
        long start = System.currentTimeMillis();

        logger.info("→ Entering  " + className + "." + methodName + "()");
        try {
            Object result = context.proceed(); // actual call
            long elapsed = System.currentTimeMillis() - start;
            logger.info("← Returning " + className + "." + methodName
                + "() [" + elapsed + " ms]");
            return result;
        } catch (Exception e) {
            logger.warning("✗ Exception in " + className + "." + methodName
                + "(): " + e.getMessage());
            throw e;
        }
    }
}


// 3. Usage — annotate the method or class
@ApplicationScoped
@Logged
public class TransferService {

    public void transferFunds(String fromId, String toId, BigDecimal amount) {
        // ... business logic — logging added automatically
    }
}

@Transactional Transaction

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.transaction.Transactional;
import jakarta.transaction.Transactional.TxType;
import java.math.BigDecimal;

@ApplicationScoped
public class BankingService {

    @Inject private AccountRepository accountRepository;
    @Inject private AuditService auditService;

    // REQUIRED: joins existing transaction or creates one
    @Transactional(TxType.REQUIRED)
    public void transferFunds(String fromAccountId, String toAccountId,
                              BigDecimal amount) {
        Account source = accountRepository.findById(fromAccountId)
            .orElseThrow(() -> new AccountNotFoundException(fromAccountId));
        Account target = accountRepository.findById(toAccountId)
            .orElseThrow(() -> new AccountNotFoundException(toAccountId));

        if (source.getBalance().compareTo(amount) < 0) {
            throw new InsufficientFundsException(
                "Insufficient funds: " + source.getBalance());
        }

        source.debit(amount);
        target.credit(amount);
        accountRepository.save(source);
        accountRepository.save(target);

        // Audit in an INDEPENDENT transaction (always persisted even on rollback)
        auditService.auditLog("TRANSFER", fromAccountId, toAccountId, amount);
    }

    // REQUIRES_NEW: independent transaction — survives caller rollback
    @Transactional(TxType.REQUIRES_NEW)
    public void auditLog(String operation, String from, String to, BigDecimal amount) {
        AuditEntry entry = new AuditEntry(operation, from, to, amount);
        try {
            auditRepository.save(entry);
        } catch (Exception e) {
            // Explicit rollback of this transaction only
            jakarta.transaction.TransactionSynchronizationRegistry tsr =
                CDI.current().select(
                    jakarta.transaction.TransactionSynchronizationRegistry.class).get();
            tsr.setRollbackOnly();
            throw e;
        }
    }

    // MANDATORY: requires an active transaction — exception otherwise
    @Transactional(TxType.MANDATORY)
    public void updateBalance(Account account) {
        accountRepository.save(account);
    }
}

CDI Events

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.event.Event;
import jakarta.enterprise.event.Observes;
import jakarta.enterprise.event.ObservesAsync;
import jakarta.inject.Inject;

// 1. Event class (POJO)
public class ClientCreatedEvent {
    private final String clientId;
    private final String email;

    public ClientCreatedEvent(String clientId, String email) {
        this.clientId = clientId;
        this.email    = email;
    }
    public String getClientId() { return clientId; }
    public String getEmail()    { return email; }
}


// 2. Event producer
@ApplicationScoped
public class EventPublisher {

    @Inject
    private Event<ClientCreatedEvent> clientCreatedEvent;

    public Client createClient(String name, String email) {
        Client client = new Client(name, email);
        clientRepository.save(client);

        // Synchronous firing — all @Observes observers execute before return
        clientCreatedEvent.fire(new ClientCreatedEvent(client.getId(), email));

        // Or asynchronous firing — non-blocking, calls @ObservesAsync observers
        // clientCreatedEvent.fireAsync(new ClientCreatedEvent(client.getId(), email));

        return client;
    }
}


// 3. Observers
@ApplicationScoped
public class NotificationObserver {

    // Synchronous observer — same thread as fire()
    public void onClientCreated(@Observes ClientCreatedEvent event) {
        System.out.println("Sending welcome email to: " + event.getEmail());
        emailService.sendWelcome(event.getEmail());
    }
}

@ApplicationScoped
public class AuditObserver {

    // Asynchronous observer — separate thread, triggered by fireAsync()
    public void onClientCreatedAsync(@ObservesAsync ClientCreatedEvent event) {
        auditService.log("NEW_CLIENT", event.getClientId());
    }
}
