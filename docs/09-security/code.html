Custom Authentication Mechanism

java 
@ApplicationScoped
@AutoApplySession
@RememberMe(cookieMaxAgeSeconds = 86400)
public class JwtAuthenticationMechanism implements HttpAuthenticationMechanism {

    @Inject
    private JwtUtil jwtUtil;

    @Override
    public AuthenticationStatus validateRequest(
            HttpServletRequest request,
            HttpServletResponse response,
            HttpMessageContext context) throws AuthenticationException {

        String authHeader = request.getHeader("Authorization");

        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            return context.responseUnauthorized();
        }

        String token = authHeader.substring(7);

        try {
            Claims claims = jwtUtil.validateToken(token);
            String username = claims.getSubject();
            Set<String> groups = new HashSet<>(
                claims.get("groups", List.class));

            CallerPrincipalCallback callerCallback =
                new CallerPrincipalCallback(
                    context.getClientSubject(), username);

            GroupPrincipalCallback groupCallback =
                new GroupPrincipalCallback(
                    context.getClientSubject(),
                    groups.toArray(new String[0]));

            context.getHandler().handle(
                new Callback[]{callerCallback, groupCallback});

            return context.notifyContainerAboutLogin(
                callerCallback.getPrincipal(), groups);

        } catch (JwtException e) {
            return context.responseUnauthorized();
        }
    }
}

IdentityStore with database

java 
@ApplicationScoped
public class DatabaseIdentityStore implements IdentityStore {

    @Inject
    private EntityManager em;

    @Inject
    private PasswordHashService passwordHashService;

    @Override
    public CredentialValidationResult validate(Credential credential) {
        if (!(credential instanceof UsernamePasswordCredential)) {
            return CredentialValidationResult.NOT_VALIDATED_RESULT;
        }
        UsernamePasswordCredential upc =
            (UsernamePasswordCredential) credential;

        try {
            AppUser user = em.createQuery(
                "SELECT u FROM AppUser u WHERE u.username = :username",
                AppUser.class)
                .setParameter("username", upc.getCaller())
                .getSingleResult();

            if (passwordHashService.verify(
                    upc.getPasswordAsString(), user.getPasswordHash())) {
                Set<String> groups = getCallerGroups(
                    new CredentialValidationResult(user.getUsername()));
                return new CredentialValidationResult(
                    user.getUsername(), groups);
            }
        } catch (NoResultException e) {
            // User not found — same delay to prevent timing attack
            passwordHashService.dummyVerify();
        }
        return CredentialValidationResult.INVALID_RESULT;
    }

    @Override
    public Set<String> getCallerGroups(CredentialValidationResult result) {
        return em.createQuery(
            "SELECT r.name FROM AppUser u JOIN u.roles r " +
            "WHERE u.username = :username", String.class)
            .setParameter("username", result.getCallerPrincipal().getName())
            .getResultStream()
            .collect(Collectors.toSet());
    }
}

Custom IdentityStore (Jakarta Security)

Jakarta Security 3.0 provides the IdentityStore API for pluggable credential validation and group retrieval, replacing vendor-specific JAAS LoginModules.

java 
@ApplicationScoped
public class CustomIdentityStore implements IdentityStore {

    @Inject
    private UserRepository userRepository;

    @Override
    public CredentialValidationResult validate(UsernamePasswordCredential credential) {
        User user = userRepository.findByUsername(credential.getCaller());
        if (user != null && user.passwordMatches(credential.getPasswordAsString())) {
            return new CredentialValidationResult(user.getUsername(),
                new HashSet<>(user.getRoles()));
        }
        return CredentialValidationResult.INVALID_RESULT;
    }
}

This custom store integrates with any data source (JPA, LDAP, etc.) and is automatically discovered by the Jakarta Security runtime via CDI.

Jakarta Security API — SecurityContext

java 
@Path("/accounts")
@Produces(MediaType.APPLICATION_JSON)
@RolesAllowed({"USER", "ADMIN"})
public class AccountResource {

    @Inject
    private SecurityContext securityContext;

    @Inject
    private AccountService accountService;

    @GET
    public Response getAccounts() {
        // Retrieve the identity of the logged-in user
        Principal caller = securityContext.getCallerPrincipal();
        String username = caller.getName();

        // Programmatic role check
        if (securityContext.isCallerInRole("ADMIN")) {
            // ADMIN sees all accounts
            return Response.ok(accountService.findAll()).build();
        } else {
            // USER only sees their own accounts
            return Response.ok(
                accountService.findByOwner(username)).build();
        }
    }

    @DELETE
    @Path("/{id}")
    public Response deleteAccount(@PathParam("id") Long id) {
        // Web resource access check (programmatic)
        if (!securityContext.hasAccessToWebResource("/admin", "GET")) {
            return Response.status(Response.Status.FORBIDDEN)
                .entity("Only administrators can delete accounts")
                .build();
        }
        accountService.delete(id);
        return Response.noContent().build();
    }
}

MicroProfile JWT — Configuration and Claims

Tip: MicroProfile JWT Propagation (MP-JWT) standardizes JWT-based authentication across microservices. Annotate your JAX-RS application with @LoginConfig(authMethod="MP-JWT") and inject claims with @Claim.

java 
// Enable MP-JWT on the application
@LoginConfig(authMethod = "MP-JWT", realmName = "banking-realm")
@ApplicationPath("/api")
public class SecureApp extends Application { }

// -------------------------------------------------------

@Path("/accounts")
@RequestScoped
@RolesAllowed("USER")
public class AccountResource {

    // Direct injection of JWT claims via CDI
    @Inject
    @Claim("sub")
    private String subject;           // User identifier

    @Inject
    @Claim("groups")
    private Set<String> groups;       // User roles/groups

    @Inject
    @Claim("upn")
    private String userPrincipalName; // Email or principal identifier

    @Inject
    @Claim("exp")
    private Long expiration;          // Token expiration timestamp

    @GET
    @Produces(MediaType.APPLICATION_JSON)
    public Response getMyAccounts() {
        return Response.ok(Map.of(
            "owner", subject,
            "groups", groups,
            "upn", userPrincipalName
        )).build();
    }
}

                  host="*"
                  httpPort="-1"

</server>

JWT Generation and Validation

java 
@ApplicationScoped
public class JwtUtil {

    @Inject
    @ConfigProperty(name = "jwt.issuer", 
                    defaultValue = "https://auth.banking.esipe.fr")
    private String issuer;

    private PrivateKey privateKey;
    private PublicKey publicKey;

    @PostConstruct
    public void init() throws Exception {
        // Load keys from resources or keystore
        this.privateKey = loadPrivateKey();
        this.publicKey  = loadPublicKey();
    }

    /**
     * Generates a JWT token signed with RS256
     */
    public String generateToken(String username, Set<String> roles) {
        Date now        = new Date();
        Date expiration = new Date(now.getTime() + 3600_000L); // 1h

        return Jwts.builder()
            .subject(username)
            .issuer(issuer)
            .audience().add("banking-api").and()
            .issuedAt(now)
            .expiration(expiration)
            .claim("groups", new ArrayList<>(roles))
            .claim("upn", username)
            .signWith(privateKey, Jwts.SIG.RS256)
            .compact();
    }

    /**
     * Validates a JWT token and returns the claims
     */
    public Claims validateToken(String token) {
        return Jwts.parser()
            .verifyWith(publicKey)
            .requireIssuer(issuer)
            .build()
            .parseSignedClaims(token)
            .getPayload();
    }
}

CORS Filter for JAX-RS

java 
@Provider
@PreMatching
public class CorsFilter implements ContainerRequestFilter,
                                   ContainerResponseFilter {

    private static final Set<String> ALLOWED_ORIGINS = Set.of(
        "https://banking.esipe.fr",
        "https://app.banking.esipe.fr"
    );

    // Intercepts BEFORE routing — handles OPTIONS preflight
    @Override
    public void filter(ContainerRequestContext request) throws IOException {
        String method = request.getMethod();
        String origin = request.getHeaderString("Origin");

        if ("OPTIONS".equalsIgnoreCase(method) && origin != null) {
            Response.ResponseBuilder rb = Response.ok();
            buildCorsHeaders(rb, origin);
            request.abortWith(rb.build());
        }
    }

    // Adds CORS headers to all responses
    @Override
    public void filter(ContainerRequestContext request,
                       ContainerResponseContext response) throws IOException {
        String origin = request.getHeaderString("Origin");
        if (origin != null) {
            MultivaluedMap<String, Object> headers = response.getHeaders();
            if (ALLOWED_ORIGINS.contains(origin)) {
                headers.add("Access-Control-Allow-Origin", origin);
                headers.add("Vary", "Origin");
            }
            headers.add("Access-Control-Allow-Methods",
                "GET, POST, PUT, DELETE, OPTIONS, PATCH");
            headers.add("Access-Control-Allow-Headers",
                "Content-Type, Authorization, X-CSRF-Token");
            headers.add("Access-Control-Allow-Credentials", "true");
            headers.add("Access-Control-Max-Age", "3600");
        }
    }

    private void buildCorsHeaders(Response.ResponseBuilder rb, String origin) {
        if (ALLOWED_ORIGINS.contains(origin)) {
            rb.header("Access-Control-Allow-Origin", origin)
              .header("Vary", "Origin");
        }
        rb.header("Access-Control-Allow-Methods",
                  "GET, POST, PUT, DELETE, OPTIONS, PATCH")
          .header("Access-Control-Allow-Headers",
                  "Content-Type, Authorization, X-CSRF-Token")
          .header("Access-Control-Allow-Credentials", "true")
          .header("Access-Control-Max-Age", "3600");
    }
}

CSRF Protection in Servlets

java 
@WebFilter("/*")
public class CsrfFilter implements Filter {

    private static final String CSRF_TOKEN_ATTR = "CSRF_TOKEN";
    private static final String CSRF_HEADER     = "X-CSRF-Token";

    @Override
    public void doFilter(ServletRequest req, ServletResponse res,
                         FilterChain chain)
            throws IOException, ServletException {

        HttpServletRequest  request  = (HttpServletRequest) req;
        HttpServletResponse response = (HttpServletResponse) res;
        HttpSession session = request.getSession(true);

        String method = request.getMethod().toUpperCase();

        if ("GET".equals(method) || "HEAD".equals(method)) {
            // Generate the CSRF token on first visit (GET)
            if (session.getAttribute(CSRF_TOKEN_ATTR) == null) {
                String token = generateSecureToken();
                session.setAttribute(CSRF_TOKEN_ATTR, token);
                // Expose the token in a response header for SPAs
                response.setHeader(CSRF_HEADER, token);
            }
            chain.doFilter(request, response);

        } else {
            // Validate the CSRF token on mutating requests (POST, PUT, DELETE)
            String sessionToken = (String) session.getAttribute(CSRF_TOKEN_ATTR);
            String requestToken = request.getHeader(CSRF_HEADER);

            if (sessionToken == null || requestToken == null
                    || !isTokenValid(sessionToken, requestToken)) {
                response.sendError(HttpServletResponse.SC_FORBIDDEN,
                    "CSRF token validation failed");
                return;
            }
            chain.doFilter(request, response);
        }
    }

    private String generateSecureToken() {
        byte[] bytes = new byte[32];
        new SecureRandom().nextBytes(bytes);
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
    }

    /**
     * Constant-time comparison to prevent timing attacks.
     * MessageDigest.isEqual() guarantees uniform execution time.
     */
    private boolean isTokenValid(String expected, String actual) {
        byte[] expectedBytes = expected.getBytes(StandardCharsets.UTF_8);
        byte[] actualBytes   = actual.getBytes(StandardCharsets.UTF_8);
        return MessageDigest.isEqual(expectedBytes, actualBytes);
    }
}
