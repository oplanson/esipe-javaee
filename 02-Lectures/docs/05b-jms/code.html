Producteur JMS avec JMSContext

java 
import jakarta.annotation.Resource;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.jms.ConnectionFactory;
import jakarta.jms.JMSConnectionFactory;
import jakarta.jms.JMSContext;
import jakarta.jms.Queue;
import jakarta.jms.TextMessage;
import java.math.BigDecimal;

@ApplicationScoped
public class TransferEventProducer {

    // Injection du JMSContext géré par le conteneur
    @Inject
    @JMSConnectionFactory("jms/transferCF")
    private JMSContext jmsContext;

    // Injection de la Queue via JNDI
    @Resource(lookup = "jms/transferQueue")
    private Queue transferQueue;

    public void sendTransferRequest(String fromAccountId,
                                    String toAccountId,
                                    BigDecimal amount,
                                    String transferType) {
        // Construire le corps JSON du message
        String body = String.format(
            "{\"fromAccountId\":\"%s\",\"toAccountId\":\"%s\","
            + "\"amount\":%s,\"transferType\":\"%s\","
            + "\"timestamp\":%d}",
            fromAccountId, toAccountId,
            amount.toPlainString(), transferType,
            System.currentTimeMillis()
        );

        // Créer et envoyer le TextMessage
        TextMessage message = jmsContext.createTextMessage(body);
        message.setStringProperty("transferType", transferType);
        message.setDoubleProperty("amount", amount.doubleValue());

        jmsContext.createProducer()
                  .setProperty("correlationId", fromAccountId + "-" + System.nanoTime())
                  .send(transferQueue, message);

        System.out.println("Transfer request sent: " + transferType
            + " — " + amount + " from " + fromAccountId);
    }
}

Consommateur Asynchrone (MessageListener)

java 
import jakarta.annotation.Resource;
import jakarta.jms.ConnectionFactory;
import jakarta.jms.JMSException;
import jakarta.jms.Message;
import jakarta.jms.MessageConsumer;
import jakarta.jms.MessageListener;
import jakarta.jms.Queue;
import jakarta.jms.Session;
import jakarta.jms.TextMessage;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;

@WebServlet(urlPatterns = "/transfer-listener", loadOnStartup = 1)
public class TransferListenerServlet extends HttpServlet {

    @Resource(lookup = "jms/transferCF")
    private ConnectionFactory connectionFactory;

    @Resource(lookup = "jms/transferQueue")
    private Queue transferQueue;

    @Override
    public void init() throws ServletException {
        try {
            var connection = connectionFactory.createConnection();
            var session = connection.createSession(false, Session.AUTO_ACKNOWLEDGE);
            MessageConsumer consumer = session.createConsumer(transferQueue);

            // Enregistrement du listener asynchrone
            consumer.setMessageListener(new MessageListener() {
                @Override
                public void onMessage(Message msg) {
                    if (msg instanceof TextMessage textMsg) {
                        try {
                            String body = textMsg.getText();
                            String transferType = msg.getStringProperty("transferType");
                            System.out.println("Received [" + transferType + "]: " + body);
                            // Traitement applicatif ici...
                        } catch (JMSException e) {
                            throw new RuntimeException("Failed to read JMS message", e);
                        }
                    }
                }
            });

            connection.start();
        } catch (Exception e) {
            throw new ServletException("JMS listener init failed", e);
        }
    }
}

Message-Driven Bean — Queue

java 
import jakarta.ejb.ActivationConfigProperty;
import jakarta.ejb.MessageDriven;
import jakarta.inject.Inject;
import jakarta.jms.JMSException;
import jakarta.jms.Message;
import jakarta.jms.MessageListener;
import jakarta.jms.TextMessage;

@MessageDriven(activationConfig = {
    @ActivationConfigProperty(
        propertyName  = "destinationType",
        propertyValue = "jakarta.jms.Queue"
    ),
    @ActivationConfigProperty(
        propertyName  = "destination",
        propertyValue = "jms/transferQueue"
    ),
    @ActivationConfigProperty(
        propertyName  = "acknowledgeMode",
        propertyValue = "Auto-acknowledge"
    )
})
public class TransferProcessorMDB implements MessageListener {

    @Inject
    private BankingService bankingService;

    // onMessage() s'exécute dans une transaction CMT gérée par le conteneur.
    // En cas d'exception RuntimeException, la transaction est rollbackée
    // et le message est rélivré par le broker.
    @Override
    public void onMessage(Message message) {
        if (!(message instanceof TextMessage textMsg)) {
            return; // ignorer les types inconnus
        }
        try {
            String body         = textMsg.getText();
            String transferType = message.getStringProperty("transferType");

            System.out.println("Processing transfer [" + transferType + "]: " + body);

            // Déléguer au service métier — dans la même transaction CMT
            bankingService.processTransfer(body);

        } catch (JMSException e) {
            // Propager comme RuntimeException pour déclencher le rollback CMT
            throw new RuntimeException("Failed to read transfer message", e);
        }
    }
}

MDB avec Topic (Pub/Sub)

java 
import jakarta.ejb.ActivationConfigProperty;
import jakarta.ejb.MessageDriven;
import jakarta.inject.Inject;
import jakarta.jms.JMSException;
import jakarta.jms.Message;
import jakarta.jms.MessageListener;
import jakarta.jms.TextMessage;

@MessageDriven(activationConfig = {
    @ActivationConfigProperty(
        propertyName  = "destinationType",
        propertyValue = "jakarta.jms.Topic"
    ),
    @ActivationConfigProperty(
        propertyName  = "destination",
        propertyValue = "jms/accountEventsTopic"
    ),
    // Abonnement durable — reçoit les messages même si le MDB était hors ligne
    @ActivationConfigProperty(
        propertyName  = "subscriptionDurability",
        propertyValue = "Durable"
    ),
    @ActivationConfigProperty(
        propertyName  = "durableSubscriptionName",
        propertyValue = "notificationMDB-durable-sub"
    ),
    @ActivationConfigProperty(
        propertyName  = "clientId",
        propertyValue = "notificationMDBClient"
    )
})
public class NotificationMDB implements MessageListener {

    @Inject
    private NotificationService notificationService;

    @Override
    public void onMessage(Message message) {
        if (!(message instanceof TextMessage textMsg)) {
            return;
        }
        try {
            String body      = textMsg.getText();
            String eventType = message.getStringProperty("eventType");

            System.out.println("Topic event received [" + eventType + "]: " + body);

            // Tous les abonnés actifs reçoivent ce message
            notificationService.dispatch(eventType, body);

        } catch (JMSException e) {
            throw new RuntimeException("Failed to read topic message", e);
        }
    }
}

Dead Letter Queue Handler

java 
import jakarta.ejb.ActivationConfigProperty;
import jakarta.ejb.MessageDriven;
import jakarta.inject.Inject;
import jakarta.jms.JMSException;
import jakarta.jms.Message;
import jakarta.jms.MessageListener;
import jakarta.jms.TextMessage;
import java.util.logging.Level;
import java.util.logging.Logger;

@MessageDriven(activationConfig = {
    @ActivationConfigProperty(
        propertyName  = "destinationType",
        propertyValue = "jakarta.jms.Queue"
    ),
    @ActivationConfigProperty(
        propertyName  = "destination",
        propertyValue = "jms/transferDLQ"
    )
})
public class TransferDLQHandlerMDB implements MessageListener {

    private static final Logger logger =
        Logger.getLogger(TransferDLQHandlerMDB.class.getName());

    @Inject
    private AlertService alertService;

    @Inject
    private DeadLetterArchiveService archiveService;

    @Override
    public void onMessage(Message message) {
        try {
            String messageId      = message.getJMSMessageID();
            int    deliveryCount  = message.getIntProperty("JMSXDeliveryCount");
            String body           = message instanceof TextMessage t ? t.getText() : "N/A";

            logger.log(Level.SEVERE,
                "DEAD LETTER: messageId={0}, deliveryCount={1}, body={2}",
                new Object[]{ messageId, deliveryCount, body });

            // Alerter l'équipe opérationnelle
            alertService.sendCriticalAlert(
                "Transfer message in DLQ",
                "messageId=" + messageId + " | deliveryCount=" + deliveryCount
                + "\nbody=" + body
            );

            // Archiver pour analyse et rejoue manuelle éventuelle
            archiveService.archive(messageId, body, deliveryCount);

        } catch (JMSException e) {
            logger.log(Level.SEVERE, "Failed to process DLQ message", e);
            // Ne pas propager — éviter une boucle infinie sur la DLQ elle-même
        }
    }
}

Sélecteur de Messages

java 
import jakarta.ejb.ActivationConfigProperty;
import jakarta.ejb.MessageDriven;
import jakarta.inject.Inject;
import jakarta.jms.JMSException;
import jakarta.jms.Message;
import jakarta.jms.MessageListener;
import jakarta.jms.TextMessage;

// Ce MDB ne consomme QUE les messages dont la propriété
// transferType vaut 'INTERNATIONAL' — filtrage côté broker
@MessageDriven(activationConfig = {
    @ActivationConfigProperty(
        propertyName  = "destinationType",
        propertyValue = "jakarta.jms.Queue"
    ),
    @ActivationConfigProperty(
        propertyName  = "destination",
        propertyValue = "jms/transferQueue"
    ),
    // Sélecteur JMS — syntaxe SQL WHERE sur les propriétés du message
    @ActivationConfigProperty(
        propertyName  = "messageSelector",
        propertyValue = "transferType = 'INTERNATIONAL'"
    )
})
public class InternationalTransferMDB implements MessageListener {

    @Inject
    private ComplianceService complianceService;

    @Override
    public void onMessage(Message message) {
        if (!(message instanceof TextMessage textMsg)) {
            return;
        }
        try {
            String body = textMsg.getText();
            System.out.println("International transfer received: " + body);

            // Traitement spécifique aux virements internationaux :
            // vérification SWIFT, contrôle OFAC, rapport réglementaire
            complianceService.processInternationalTransfer(body);

        } catch (JMSException e) {
            throw new RuntimeException("Failed to read international transfer message", e);
        }
    }
}

Configuration Liberty JMS

xml 
<?xml version="1.0" encoding="UTF-8"?>
<server description="JMS Banking Application">

  <!-- Features JMS / MDB nécessaires -->
  <featureManager>
    <feature>jakartaee-10.0</feature>
    <feature>messagingClient-3.0</feature>
    <feature>mdb-4.0</feature>
  </featureManager>

  <!-- Connexion au broker JMS Liberty Messaging -->
  <wasJmsEndpoint host="localhost" port="7276"
                  wasJmsPort="7276"/>

  <!-- ConnectionFactory injectée via @JMSConnectionFactory("jms/transferCF") -->
  <jmsConnectionFactory id="transferCF" jndiName="jms/transferCF">
    <properties.wasJms/>
  </jmsConnectionFactory>

  <!-- Queue pour les transferts — Point-à-Point -->
  <jmsQueue id="transferQueue" jndiName="jms/transferQueue">
    <properties.wasJms queueName="TRANSFER.QUEUE"/>
  </jmsQueue>

  <!-- Dead Letter Queue -->
  <jmsQueue id="transferDLQ" jndiName="jms/transferDLQ">
    <properties.wasJms queueName="TRANSFER.DLQ"/>
  </jmsQueue>

  <!-- Topic pour les événements de compte — Pub/Sub -->
  <jmsTopic id="accountEventsTopic" jndiName="jms/accountEventsTopic">
    <properties.wasJms topicName="ACCOUNT.EVENTS"/>
  </jmsTopic>

  <!-- ActivationSpec pour le MDB TransferProcessorMDB -->
  <jmsActivationSpec id="BankingApp/TransferProcessorMDB">
    <properties.wasJms
        destinationRef="transferQueue"
        destinationType="jakarta.jms.Queue"
        maxConcurrentEndpoints="10"/>
  </jmsActivationSpec>

  <!-- ActivationSpec pour le DLQ Handler MDB -->
  <jmsActivationSpec id="BankingApp/TransferDLQHandlerMDB">
    <properties.wasJms
        destinationRef="transferDLQ"
        destinationType="jakarta.jms.Queue"
        maxConcurrentEndpoints="2"/>
  </jmsActivationSpec>

  <!-- ActivationSpec pour le MDB Topic (abonnement durable) -->
  <jmsActivationSpec id="BankingApp/NotificationMDB">
    <properties.wasJms
        destinationRef="accountEventsTopic"
        destinationType="jakarta.jms.Topic"
        durableSubscriptionHome="notificationMDBClient"
        subscriptionDurability="Durable"/>
  </jmsActivationSpec>

</server>
