Primary Port — Use Case Interface

java 
// Package : com.bank.domain.transfer
// Port primaire — interface use case dans le DOMAINE (aucune annotation Jakarta EE)

package com.bank.domain.transfer;

import com.bank.domain.account.AccountId;
import com.bank.domain.shared.Money;
import java.util.UUID;

/**
 * Port primaire : expose le use case de virement monétaire.
 * Implémenté par MoneyTransferService dans la couche application.
 * Appelé par TransferRestAdapter dans la couche infrastructure.
 */
public interface MoneyTransferUseCase {

    TransferResult transferMoney(TransferCommand command);

}

// Commande immuable — record Java 17
record TransferCommand(
    AccountId fromAccountId,
    AccountId toAccountId,
    Money amount,
    UUID requestId
) {
    // Validation dans le constructeur compact
    public TransferCommand {
        if (amount == null || amount.isNegativeOrZero()) {
            throw new IllegalArgumentException("Transfer amount must be positive");
        }
        if (fromAccountId.equals(toAccountId)) {
            throw new IllegalArgumentException("Cannot transfer to the same account");
        }
    }
}

// Résultat immuable — record Java 17
record TransferResult(
    UUID transferId,
    String status,
    Money amountTransferred
) {}

Secondary Port — Repository Interface

java 
// Package : com.bank.domain.account
// Port secondaire — interface repository dans le DOMAINE (aucune annotation JPA)

package com.bank.domain.account;

import com.bank.domain.client.ClientId;
import java.util.List;
import java.util.Optional;

/**
 * Port secondaire : contrat d'accès aux comptes.
 * Défini dans le domaine — implémenté par JpaAccountAdapter dans l'infrastructure.
 * Le domaine ne sait PAS si la persistance est JPA, MongoDB ou in-memory.
 */
public interface AccountPort {

    /**
     * Retrouve un compte par son identifiant.
     */
    Optional<Account> findById(AccountId accountId);

    /**
     * Retrouve tous les comptes d'un client.
     */
    List<Account> findByClientId(ClientId clientId);

    /**
     * Sauvegarde un compte (création ou mise à jour).
     */
    Account save(Account account);

    /**
     * Vérifie l'existence d'un compte.
     */
    boolean exists(AccountId accountId);

}

Primary Adapter — REST Resource

java 
// Package : com.bank.infrastructure.web
// Adapter primaire — ressource JAX-RS dans l'INFRASTRUCTURE

package com.bank.infrastructure.web;

import com.bank.domain.account.AccountId;
import com.bank.domain.shared.Money;
import com.bank.domain.transfer.MoneyTransferUseCase;
import com.bank.domain.transfer.TransferCommand;
import com.bank.domain.transfer.TransferResult;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.*;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import java.math.BigDecimal;
import java.util.UUID;

@Path("/transfers")
@ApplicationScoped
@Produces(MediaType.APPLICATION_JSON)
@Consumes(MediaType.APPLICATION_JSON)
public class TransferRestAdapter {

    // Injection du PORT PRIMAIRE (interface domaine) — pas de l'implémentation concrète
    @Inject
    private MoneyTransferUseCase moneyTransferUseCase;

    @POST
    public Response transferMoney(TransferRequest dto) {
        // Mapping DTO → Command domaine
        TransferCommand command = toCommand(dto);

        // Délégation au port primaire — l'adapter ne contient PAS de logique métier
        TransferResult result = moneyTransferUseCase.transferMoney(command);

        return Response.ok(result).build();
    }

    // Mapping DTO HTTP → Command domaine — responsabilité de l'adapter
    private TransferCommand toCommand(TransferRequest dto) {
        return new TransferCommand(
            new AccountId(dto.fromAccountId()),
            new AccountId(dto.toAccountId()),
            Money.of(dto.amount(), dto.currency()),
            UUID.randomUUID()
        );
    }

}

// DTO d'entrée REST — record Java 17
record TransferRequest(
    String fromAccountId,
    String toAccountId,
    BigDecimal amount,
    String currency
) {}

Secondary Adapter — JPA Repository

java 
// Package : com.bank.infrastructure.persistence
// Adapter secondaire — implémentation JPA du port AccountPort dans l'INFRASTRUCTURE

package com.bank.infrastructure.persistence;

import com.bank.domain.account.Account;
import com.bank.domain.account.AccountId;
import com.bank.domain.account.AccountPort;
import com.bank.domain.client.ClientId;
import com.bank.domain.shared.Money;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import jakarta.persistence.TypedQuery;
import java.util.List;
import java.util.Optional;

@ApplicationScoped
public class JpaAccountAdapter implements AccountPort {

    // EntityManager JPA — uniquement dans l'infrastructure, jamais dans le domaine
    @PersistenceContext
    private EntityManager entityManager;

    @Override
    public Optional<Account> findById(AccountId accountId) {
        AccountJpaEntity entity = entityManager.find(AccountJpaEntity.class, accountId.value());
        return Optional.ofNullable(entity).map(this::toDomain);
    }

    @Override
    public List<Account> findByClientId(ClientId clientId) {
        TypedQuery<AccountJpaEntity> query = entityManager.createNamedQuery(
            "Account.findByClientId", AccountJpaEntity.class
        );
        query.setParameter("clientId", clientId.value());
        return query.getResultList().stream().map(this::toDomain).toList();
    }

    @Override
    public Account save(Account account) {
        AccountJpaEntity entity = toJpa(account);
        AccountJpaEntity saved = entityManager.merge(entity);
        return toDomain(saved);
    }

    @Override
    public boolean exists(AccountId accountId) {
        return entityManager.find(AccountJpaEntity.class, accountId.value()) != null;
    }

    // Mapping entité JPA → entité domaine
    private Account toDomain(AccountJpaEntity e) {
        return new Account(
            new AccountId(e.getId()),
            new ClientId(e.getClientId()),
            Money.of(e.getBalance(), e.getCurrency()),
            e.getStatus()
        );
    }

    // Mapping entité domaine → entité JPA
    private AccountJpaEntity toJpa(Account account) {
        AccountJpaEntity e = new AccountJpaEntity();
        e.setId(account.id().value());
        e.setClientId(account.clientId().value());
        e.setBalance(account.balance().amount());
        e.setCurrency(account.balance().currency());
        e.setStatus(account.status());
        return e;
    }

}

Application Service — Use Case

java 
// Package : com.bank.application
// Service applicatif — implémente le port primaire MoneyTransferUseCase
// Orchestre le domaine et les ports secondaires

package com.bank.application;

import com.bank.domain.account.Account;
import com.bank.domain.account.AccountId;
import com.bank.domain.account.AccountPort;
import com.bank.domain.transfer.MoneyTransferUseCase;
import com.bank.domain.transfer.TransferCommand;
import com.bank.domain.transfer.TransferResult;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.transaction.Transactional;
import java.util.UUID;

@ApplicationScoped
public class MoneyTransferService implements MoneyTransferUseCase {

    // Injection du PORT SECONDAIRE (interface domaine AccountPort)
    // CDI injecte JpaAccountAdapter à l'exécution — le service ne connaît pas JPA
    @Inject
    private AccountPort accountPort;

    @Override
    @Transactional
    public TransferResult transferMoney(TransferCommand command) {
        // 1. Charger les entités via le port secondaire
        Account source = accountPort.findById(command.fromAccountId())
            .orElseThrow(() -> new AccountNotFoundException(command.fromAccountId()));

        Account target = accountPort.findById(command.toAccountId())
            .orElseThrow(() -> new AccountNotFoundException(command.toAccountId()));

        // 2. Appliquer la logique métier sur les entités domaine
        //    Les règles métier (solde suffisant, compte actif) sont dans le domaine
        source.debitAccount(command.amount());
        target.creditAccount(command.amount());

        // 3. Persister via le port secondaire
        accountPort.save(source);
        accountPort.save(target);

        // 4. Retourner le résultat
        return new TransferResult(
            UUID.randomUUID(),
            "COMPLETED",
            command.amount()
        );
    }

}

Tests with Mock Adapter

java 
// Mock adapter in-memory — implémente AccountPort sans JPA, sans base de données
package com.bank.application;

import com.bank.domain.account.Account;
import com.bank.domain.account.AccountId;
import com.bank.domain.account.AccountPort;
import com.bank.domain.client.ClientId;
import java.util.*;

/**
 * Mock adapter in-memory pour les tests unitaires.
 * Implémente AccountPort sans aucune dépendance technique.
 */
class AccountPortMock implements AccountPort {

    private final Map<AccountId, Account> store = new HashMap<>();

    // Méthode utilitaire pour pré-charger des données dans le mock
    void givenAccount(Account account) {
        store.put(account.id(), account);
    }

    @Override
    public Optional<Account> findById(AccountId accountId) {
        return Optional.ofNullable(store.get(accountId));
    }

    @Override
    public List<Account> findByClientId(ClientId clientId) {
        return store.values().stream()
            .filter(a -> a.clientId().equals(clientId))
            .toList();
    }

    @Override
    public Account save(Account account) {
        store.put(account.id(), account);
        return account;
    }

    @Override
    public boolean exists(AccountId accountId) {
        return store.containsKey(accountId);
    }
}
java 
// Test du use case — POJO pur, aucun Spring, aucun CDI, aucun serveur
package com.bank.application;

import com.bank.domain.account.Account;
import com.bank.domain.account.AccountId;
import com.bank.domain.client.ClientId;
import com.bank.domain.shared.Money;
import com.bank.domain.transfer.TransferCommand;
import com.bank.domain.transfer.TransferResult;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import java.math.BigDecimal;
import java.util.UUID;

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

class MoneyTransferServiceTest {

    private AccountPortMock accountPort;
    private MoneyTransferService service;

    @BeforeEach
    void setUp() {
        accountPort = new AccountPortMock();
        service = new MoneyTransferService();
        // Injection manuelle — pas besoin de CDI dans les tests unitaires
        service.accountPort = accountPort;
    }

    @Test
    void shouldTransferMoneyBetweenAccounts() {
        // GIVEN
        AccountId sourceId = new AccountId("ACC-001");
        AccountId targetId = new AccountId("ACC-002");
        Money initialBalance = Money.of(new BigDecimal("500.00"), "EUR");
        Money transferAmount = Money.of(new BigDecimal("100.00"), "EUR");

        accountPort.givenAccount(Account.open(sourceId, new ClientId("CLI-1"), initialBalance));
        accountPort.givenAccount(Account.open(targetId, new ClientId("CLI-2"), initialBalance));

        TransferCommand command = new TransferCommand(
            sourceId, targetId, transferAmount, UUID.randomUUID()
        );

        // WHEN
        TransferResult result = service.transferMoney(command);

        // THEN
        assertThat(result.status()).isEqualTo("COMPLETED");
        assertThat(accountPort.findById(sourceId).get().balance())
            .isEqualTo(Money.of(new BigDecimal("400.00"), "EUR"));
        assertThat(accountPort.findById(targetId).get().balance())
            .isEqualTo(Money.of(new BigDecimal("600.00"), "EUR"));
    }

    @Test
    void shouldRejectTransferWhenInsufficientBalance() {
        // GIVEN
        AccountId sourceId = new AccountId("ACC-003");
        AccountId targetId = new AccountId("ACC-004");
        Money smallBalance = Money.of(new BigDecimal("10.00"), "EUR");
        Money largeAmount  = Money.of(new BigDecimal("500.00"), "EUR");

        accountPort.givenAccount(Account.open(sourceId, new ClientId("CLI-3"), smallBalance));
        accountPort.givenAccount(Account.open(targetId, new ClientId("CLI-4"), smallBalance));

        TransferCommand command = new TransferCommand(
            sourceId, targetId, largeAmount, UUID.randomUUID()
        );

        // THEN — exception métier levée par le domaine
        assertThatThrownBy(() -> service.transferMoney(command))
            .isInstanceOf(InsufficientBalanceException.class);
    }

}

Testing Domain Logic Without Infrastructure

A key benefit of Hexagonal Architecture is the ability to test domain logic as plain Java objects, without starting a CDI container or database.

java 
// Plain JUnit 5 test — no @QuarkusTest, no @Inject, no database
class OrderServiceTest {

    @Test
    void shouldRejectOrderWhenStockInsufficient() {
        // Arrange — use in-memory stub, no real repository
        InMemoryProductRepository repo = new InMemoryProductRepository();
        repo.save(new Product("P1", "Widget", 2));
        OrderService service = new OrderService(repo);

        // Act
        Exception ex = assertThrows(IllegalStateException.class,
            () -> service.placeOrder("P1", 10));

        // Assert
        assertTrue(ex.getMessage().contains("insufficient stock"));
    }
}

The InMemoryProductRepository implements the port interface and keeps data in a simple Map — no JPA, no transactions needed.
