Stateless Session Bean — Banking Service

java 
import jakarta.ejb.Stateless;
import jakarta.ejb.TransactionAttribute;
import jakarta.ejb.TransactionAttributeType;
import jakarta.inject.Inject;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;

@Stateless
public class BankingServiceBean {

    @Inject
    private EntityManager em;

    // REQUIRED : participates in active transaction or creates a new one
    @TransactionAttribute(TransactionAttributeType.REQUIRED)
    public void transferFunds(String fromAccountId, String toAccountId,
                              BigDecimal amount) {
        Account source = em.find(Account.class, fromAccountId);
        Account target = em.find(Account.class, toAccountId);

        if (source == null || target == null) {
            throw new IllegalArgumentException("Account not found");
        }

        validateTransfer(source, amount);

        source.debit(amount);
        target.credit(amount);

        em.merge(source);
        em.merge(target);
    }

    // MANDATORY : must be called within an existing transaction context
    @TransactionAttribute(TransactionAttributeType.MANDATORY)
    private void validateTransfer(Account source, BigDecimal amount) {
        if (amount.compareTo(BigDecimal.ZERO) <= 0) {
            throw new IllegalArgumentException("Amount must be positive");
        }
        if (source.getBalance().compareTo(amount) < 0) {
            throw new InsufficientFundsException(
                "Insufficient funds: available=" + source.getBalance()
                + ", requested=" + amount);
        }
    }
}

Stateful Session Bean — Shopping Cart

java 
import jakarta.ejb.Remove;
import jakarta.ejb.Stateful;
import jakarta.enterprise.context.SessionScoped;
import jakarta.annotation.PreDestroy;
import java.io.Serializable;
import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.List;
import java.util.logging.Logger;

@Stateful
@SessionScoped
public class ShoppingCartBean implements Serializable {

    private static final Logger logger =
        Logger.getLogger(ShoppingCartBean.class.getName());

    // State preserved between calls from the same client
    private final List<Item> items = new ArrayList<>();

    public void addItem(Item item) {
        items.add(item);
        logger.info("Item added: " + item.getName()
            + " — cart size: " + items.size());
    }

    public boolean removeItem(String itemId) {
        return items.removeIf(i -> i.getId().equals(itemId));
    }

    public List<Item> getItems() {
        return List.copyOf(items);
    }

    public BigDecimal getTotal() {
        return items.stream()
            .map(Item::getPrice)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
    }

    public Order checkout(String customerId) {
        if (items.isEmpty()) {
            throw new EmptyCartException("Cart is empty for customer: " + customerId);
        }
        Order order = new Order(customerId, List.copyOf(items), getTotal());
        items.clear();
        logger.info("Order created: " + order.getId() + " for customer: " + customerId);
        return order;
    }

    // @Remove : releases the Stateful instance after this call — end of conversation
    @Remove
    public void clearCart() {
        items.clear();
        logger.info("Cart cleared and Stateful bean released");
    }

    @PreDestroy
    public void onDestroy() {
        logger.info("ShoppingCartBean destroyed — items remaining: " + items.size());
    }
}

Singleton — Application Configuration

java 
import jakarta.annotation.PostConstruct;
import jakarta.ejb.ConcurrencyManagement;
import jakarta.ejb.ConcurrencyManagementType;
import jakarta.ejb.Lock;
import jakarta.ejb.LockType;
import jakarta.ejb.Singleton;
import jakarta.ejb.Startup;
import java.util.concurrent.ConcurrentHashMap;
import java.util.logging.Logger;

@Singleton
@Startup  // initialised at server startup, before any client call
@ConcurrencyManagement(ConcurrencyManagementType.CONTAINER)
public class AppConfigBean {

    private static final Logger logger =
        Logger.getLogger(AppConfigBean.class.getName());

    // ConcurrentHashMap protected by container @Lock
    private final ConcurrentHashMap<String, String> config =
        new ConcurrentHashMap<>();

    @PostConstruct
    public void loadConfig() {
        // Initial configuration loading at startup
        config.put("app.name", "BankingApp");
        config.put("app.version", "3.2.1");
        config.put("max.transfer.amount", "100000");
        config.put("session.timeout.minutes", "30");
        logger.info("AppConfigBean initialized with " + config.size() + " properties");
    }

    // @Lock(WRITE) : exclusive access — only one thread at a time can write
    @Lock(LockType.WRITE)
    public void updateConfig(String key, String value) {
        String old = config.put(key, value);
        logger.info("Config updated: " + key + " = " + value
            + " (was: " + old + ")");
    }

    // @Lock(READ) : concurrent read access — multiple simultaneous threads allowed
    @Lock(LockType.READ)
    public String getConfig(String key) {
        return config.getOrDefault(key, "");
    }

    @Lock(LockType.READ)
    public String getConfig(String key, String defaultValue) {
        return config.getOrDefault(key, defaultValue);
    }
}

Message-Driven Bean

java 
import jakarta.ejb.ActivationConfigProperty;
import jakarta.ejb.MessageDriven;
import jakarta.inject.Inject;
import jakarta.jms.JMSException;
import jakarta.jms.Message;
import jakarta.jms.MessageListener;
import jakarta.jms.TextMessage;
import java.math.BigDecimal;
import java.util.logging.Logger;

@MessageDriven(activationConfig = {
    @ActivationConfigProperty(
        propertyName  = "destinationType",
        propertyValue = "jakarta.jms.Queue"),
    @ActivationConfigProperty(
        propertyName  = "destinationLookup",
        propertyValue = "jms/TransferRequestQueue"),
    @ActivationConfigProperty(
        propertyName  = "acknowledgeMode",
        propertyValue = "Auto-acknowledge")
})
public class TransferRequestMDB implements MessageListener {

    private static final Logger logger =
        Logger.getLogger(TransferRequestMDB.class.getName());

    @Inject
    private BankingServiceBean bankingService;

    @Override
    public void onMessage(Message message) {
        try {
            if (message instanceof TextMessage textMsg) {
                String payload = textMsg.getText();
                logger.info("Received transfer request: " + payload);

                // Simplified JSON payload parsing
                TransferRequest request = TransferRequest.fromJson(payload);

                // Process the wire transfer via the banking service
                bankingService.transferFunds(
                    request.getFromAccountId(),
                    request.getToAccountId(),
                    new BigDecimal(request.getAmount())
                );

                logger.info("Transfer processed successfully: "
                    + request.getFromAccountId() + " → " + request.getToAccountId()
                    + " : " + request.getAmount());
            } else {
                logger.warning("Unexpected message type: "
                    + message.getClass().getName());
            }
        } catch (JMSException e) {
            logger.severe("JMS error processing transfer request: " + e.getMessage());
            throw new RuntimeException("Failed to process transfer message", e);
        }
    }
}

Declarative Security

java 
import jakarta.annotation.Resource;
import jakarta.annotation.security.DenyAll;
import jakarta.annotation.security.PermitAll;
import jakarta.annotation.security.RolesAllowed;
import jakarta.ejb.EJBContext;
import jakarta.ejb.Stateless;

// @RolesAllowed at class level : applies to all methods by default
@Stateless
@RolesAllowed("EMPLOYEE")
public class SecureBankingService {

    @Resource
    private EJBContext ctx;

    // Override : only ADMIN can delete an account
    @RolesAllowed("ADMIN")
    public void deleteAccount(String accountId) {
        // Programmatic security: audit the admin action
        String caller = ctx.getCallerPrincipal().getName();
        auditService.log("DELETE_ACCOUNT", caller, accountId);
        accountRepository.delete(accountId);
    }

    // Override : accessible to all (authenticated or not) — public consultation
    @PermitAll
    public AccountInfo getAccountInfo(String accountId) {
        return accountRepository.findPublicInfo(accountId);
    }

    // Override : inaccessible in production — maintenance method disabled
    @DenyAll
    public void deleteAllData() {
        // This method can never be invoked via the container
        throw new UnsupportedOperationException("Not accessible");
    }

    // Inherits @RolesAllowed("EMPLOYEE") from the class
    public void updateAccountBalance(String accountId, java.math.BigDecimal amount) {
        // Complementary programmatic check based on role
        if (ctx.isCallerInRole("ADMIN")) {
            // Admins can perform unlimited adjustments
            accountRepository.forceUpdateBalance(accountId, amount);
        } else {
            // Employees are subject to standard transfer limits
            bankingService.transferFunds("INTERNAL", accountId, amount);
        }
    }
}

Timer Service

java 
import jakarta.annotation.Resource;
import jakarta.ejb.Schedule;
import jakarta.ejb.Stateless;
import jakarta.ejb.Timeout;
import jakarta.ejb.Timer;
import jakarta.ejb.TimerConfig;
import jakarta.ejb.TimerService;
import java.util.logging.Logger;

@Stateless
public class ReportSchedulerBean {

    private static final Logger logger =
        Logger.getLogger(ReportSchedulerBean.class.getName());

    @Resource
    private TimerService timerService;

    // Automatic timer : runs every day at midnight — persistent (survives server restarts)
    @Schedule(hour = "0", minute = "0", second = "0",
              persistent = true, info = "DailyReportTimer")
    public void generateDailyReport(Timer timer) {
        logger.info("Daily report generation started — timer: " + timer.getInfo());
        try {
            reportService.generateDailyReport();
            logger.info("Daily report generated successfully");
        } catch (Exception e) {
            logger.severe("Daily report generation failed: " + e.getMessage());
        }
    }

    // Automatic timer : runs every Monday at 8am for the weekly report
    @Schedule(hour = "8", minute = "0", second = "0",
              dayOfWeek = "Mon", persistent = true, info = "WeeklyReportTimer")
    public void generateWeeklyReport(Timer timer) {
        logger.info("Weekly report generation started");
        reportService.generateWeeklyReport();
    }

    // Programmatic timer creation for a delayed report
    public void scheduleDelayedReport(String reportId, long delayMs) {
        TimerConfig config = new TimerConfig(reportId, true);
        timerService.createSingleActionTimer(delayMs, config);
        logger.info("Delayed report timer created — reportId=" + reportId
            + ", delay=" + delayMs + "ms");
    }

    // @Timeout callback : called when a programmatic timer expires
    @Timeout
    public void processTimer(Timer timer) {
        String reportId = (String) timer.getInfo();
        logger.info("Processing timer callback — reportId: " + reportId);
        reportService.generateReport(reportId);
    }
}
