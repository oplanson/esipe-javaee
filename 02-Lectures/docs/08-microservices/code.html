Client REST inter-services avec Fault Tolerance

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.*;
import jakarta.ws.rs.core.MediaType;
import org.eclipse.microprofile.faulttolerance.*;
import org.eclipse.microprofile.rest.client.inject.RegisterRestClient;
import org.eclipse.microprofile.rest.client.inject.RestClient;

// Interface cliente — @RegisterRestClient avec configKey pour l'URL dans MicroProfile Config
@RegisterRestClient(configKey = "account-service")
@Path("/accounts")
@Produces(MediaType.APPLICATION_JSON)
public interface AccountServiceClient {

    @GET
    @Path("/{id}")
    Account getAccount(@PathParam("id") Long id);
}

// Modèle de réponse partagé
public record Account(Long id, String iban, double balance, String ownerId) {}


// Service applicatif avec Fault Tolerance
@ApplicationScoped
public class AccountQueryService {

    @Inject
    @RestClient
    private AccountServiceClient accountClient;

    // @Retry        — 3 tentatives supplémentaires, délai exponentiel
    // @CircuitBreaker — ouvre si 75 % d'échecs parmi les 4 dernières requêtes
    // @Timeout      — échoue si l'appel dure plus de 2 secondes
    // @Fallback     — retourne un compte par défaut si toutes les tentatives échouent
    @Retry(maxRetries = 3, delay = 200, jitter = 100)
    @CircuitBreaker(requestVolumeThreshold = 4, failureRatio = 0.75, delay = 2000)
    @Timeout(value = 2000)
    @Fallback(fallbackMethod = "getAccountFallback")
    public Account getAccount(Long id) {
        return accountClient.getAccount(id);
    }

    // Méthode de fallback — même signature que la méthode principale
    public Account getAccountFallback(Long id) {
        // Retourne un compte temporaire depuis le cache ou une valeur par défaut
        return new Account(id, "UNAVAILABLE", 0.0, "UNKNOWN");
    }
}

// Configuration dans microprofile-config.properties :
// account-service/mp-rest/url=http://account-service:8080
// account-service/mp-rest/connectTimeout=1000
// account-service/mp-rest/readTimeout=3000

API Gateway — Request Routing

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.*;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import org.eclipse.microprofile.rest.client.inject.RestClient;

@Path("/api")
@ApplicationScoped
@Produces(MediaType.APPLICATION_JSON)
@Consumes(MediaType.APPLICATION_JSON)
public class GatewayResource {

    @Inject
    @RestClient
    private AccountServiceClient accountClient;

    @Inject
    @RestClient
    private TransferServiceClient transferClient;

    @Inject
    @RestClient
    private ClientServiceClient clientServiceClient;

    // Routing GET /api/accounts/{id} → account-service
    @GET
    @Path("/accounts/{id}")
    public Response getAccount(@PathParam("id") Long id) {
        Account account = accountClient.getAccount(id);
        return Response.ok(account).build();
    }

    // Routing POST /api/transfers → transfer-service
    @POST
    @Path("/transfers")
    public Response createTransfer(TransferRequest request) {
        TransferResponse result = transferClient.createTransfer(request);
        return Response.status(Response.Status.CREATED).entity(result).build();
    }

    // Agrégation : dashboard client — appelle client-service + account-service
    @GET
    @Path("/clients/{id}/dashboard")
    public Response getClientDashboard(@PathParam("id") String clientId) {
        // Appels parallèles avec CompletableFuture pour réduire la latence
        var clientFuture = java.util.concurrent.CompletableFuture
            .supplyAsync(() -> clientServiceClient.getClient(clientId));
        var accountsFuture = java.util.concurrent.CompletableFuture
            .supplyAsync(() -> accountClient.getAccountsByOwner(clientId));

        var dashboard = java.util.concurrent.CompletableFuture
            .allOf(clientFuture, accountsFuture)
            .thenApply(v -> new ClientDashboard(
                clientFuture.join(),
                accountsFuture.join()
            ));

        return Response.ok(dashboard.join()).build();
    }
}

MicroProfile Config — Configuration multi-environnement

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import org.eclipse.microprofile.config.inject.ConfigProperty;

@ApplicationScoped
public class ServiceConfiguration {

    // URL du service de comptes — surchargée via variable d'environnement en production
    @Inject
    @ConfigProperty(name = "account.service.url",
                    defaultValue = "http://localhost:8081")
    private String accountServiceUrl;

    // URL du service de virements
    @Inject
    @ConfigProperty(name = "transfer.service.url",
                    defaultValue = "http://localhost:8082")
    private String transferServiceUrl;

    // Feature flag — active/désactive le mode debug sans recompilation
    @Inject
    @ConfigProperty(name = "feature.debug.enabled", defaultValue = "false")
    private boolean debugEnabled;

    // Seuil de limite de virement quotidien en euros
    @Inject
    @ConfigProperty(name = "transfer.daily.limit.eur", defaultValue = "10000")
    private int dailyTransferLimit;

    public String getAccountServiceUrl()  { return accountServiceUrl; }
    public String getTransferServiceUrl() { return transferServiceUrl; }
    public boolean isDebugEnabled()       { return debugEnabled; }
    public int getDailyTransferLimit()    { return dailyTransferLimit; }
}

// --- src/main/resources/META-INF/microprofile-config.properties (développement) ---
// account.service.url=http://localhost:8081
// transfer.service.url=http://localhost:8082
// feature.debug.enabled=true
// transfer.daily.limit.eur=10000

// --- Variables d'environnement Docker Compose (production) ---
// ACCOUNT_SERVICE_URL=http://account-service:8080
// TRANSFER_SERVICE_URL=http://transfer-service:8080
// FEATURE_DEBUG_ENABLED=false
// TRANSFER_DAILY_LIMIT_EUR=50000

Health Check composite

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import org.eclipse.microprofile.health.HealthCheck;
import org.eclipse.microprofile.health.HealthCheckResponse;
import org.eclipse.microprofile.health.Readiness;
import org.eclipse.microprofile.rest.client.inject.RestClient;

// @Readiness — Kubernetes retire ce pod du load balancer si ce check échoue
@Readiness
@ApplicationScoped
public class CompositeHealthCheck implements HealthCheck {

    @Inject
    private DatabaseConnectionPool dbPool;

    @Inject
    @RestClient
    private AccountServiceClient accountClient;

    @Inject
    private ServiceConfiguration config;

    @Override
    public HealthCheckResponse call() {
        boolean dbUp     = checkDatabase();
        boolean svcUp    = checkAccountService();
        boolean cfgLoaded = checkConfiguration();

        boolean allUp = dbUp && svcUp && cfgLoaded;

        return HealthCheckResponse.named("transfer-service-readiness")
            .status(allUp)
            .withData("database-connection", dbUp ? "UP" : "DOWN")
            .withData("account-service",     svcUp ? "UP" : "DOWN")
            .withData("config-loaded",       cfgLoaded ? "OK" : "MISSING")
            .withData("service-url",         config.getAccountServiceUrl())
            .build();
    }

    private boolean checkDatabase() {
        try {
            return dbPool.isValid(2); // timeout 2 secondes
        } catch (Exception e) {
            return false;
        }
    }

    private boolean checkAccountService() {
        try {
            accountClient.getAccount(0L); // appel ping — 404 attendu mais service disponible
            return true;
        } catch (jakarta.ws.rs.NotFoundException e) {
            return true;  // 404 = service up mais entité absente — c'est OK
        } catch (Exception e) {
            return false; // timeout ou connexion refusée
        }
    }

    private boolean checkConfiguration() {
        return config.getAccountServiceUrl() != null
            && !config.getAccountServiceUrl().isBlank();
    }
}

//       "config-loaded": "OK",
//       "service-url": "http://account-service:8080"

Saga — Choreography

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.event.Event;
import jakarta.enterprise.event.Observes;
import jakarta.inject.Inject;

// ---- Événements du Saga ----
public record TransferInitiatedEvent(String transferId, String fromAccount,
                                     String toAccount, double amount) {}
public record AccountDebitedEvent(String transferId, String fromAccount, double amount) {}
public record AccountCreditedEvent(String transferId, String toAccount, double amount) {}
public record TransferCompletedEvent(String transferId) {}
public record TransferFailedEvent(String transferId, String reason) {}
public record DebitCompensatedEvent(String transferId) {}


// ---- Coordinateur du Saga — Transfer Service ----
@ApplicationScoped
public class TransferSagaCoordinator {

    @Inject
    private Event<AccountDebitedEvent> debitEvent;

    @Inject
    private Event<TransferFailedEvent> failedEvent;

    @Inject
    private Event<TransferCompletedEvent> completedEvent;

    @Inject
    private TransferRepository transferRepository;

    // Étape 1 : démarrage du Saga — déclenche le débit du compte source
    public void initiateTransfer(String transferId, String from, String to, double amount) {
        transferRepository.save(new Transfer(transferId, from, to, amount, "INITIATED"));
        debitEvent.fire(new AccountDebitedEvent(transferId, from, amount));
    }

    // Étape 3 : crédit confirmé → Saga terminé avec succès
    public void onAccountCredited(@Observes AccountCreditedEvent event) {
        transferRepository.updateStatus(event.transferId(), "COMPLETED");
        completedEvent.fire(new TransferCompletedEvent(event.transferId()));
    }

    // Étape compensatoire : débit annulé → Saga en échec
    public void onDebitCompensated(@Observes DebitCompensatedEvent event) {
        transferRepository.updateStatus(event.transferId(), "FAILED");
        failedEvent.fire(new TransferFailedEvent(event.transferId(), "DEBIT_COMPENSATED"));
    }
}


// ---- Account Service — répond aux événements ----
@ApplicationScoped
public class AccountSagaParticipant {

    @Inject
    private AccountRepository accountRepository;

    @Inject
    private Event<AccountCreditedEvent> creditedEvent;

    @Inject
    private Event<DebitCompensatedEvent> compensatedEvent;

    // Étape 2a : débite le compte source
    public void onAccountDebited(@Observes AccountDebitedEvent event) {
        try {
            accountRepository.debit(event.fromAccount(), event.amount());
            // Débit réussi — fire l'événement de crédit (géré par un autre listener)
            creditedEvent.fire(new AccountCreditedEvent(
                event.transferId(), "toAccount", event.amount()));
        } catch (InsufficientFundsException e) {
            // Compensation — annuler l'opération
            compensatedEvent.fire(new DebitCompensatedEvent(event.transferId()));
        }
    }
}

Corrélation ID pour tracing

java 
import jakarta.annotation.Priority;
import jakarta.ws.rs.Priorities;
import jakarta.ws.rs.container.*;
import jakarta.ws.rs.ext.Provider;
import java.io.IOException;
import java.util.UUID;
import org.slf4j.MDC;

// Filtre JAX-RS côté serveur — s'applique à toutes les requêtes entrantes
@Provider
@Priority(Priorities.HEADER_DECORATOR)
public class CorrelationIdServerFilter implements ContainerRequestFilter,
                                                  ContainerResponseFilter {

    private static final String CORRELATION_HEADER = "X-Correlation-ID";
    private static final String MDC_KEY = "correlationId";

    // Requête entrante — extrait ou génère le correlation ID
    @Override
    public void filter(ContainerRequestContext requestContext) throws IOException {
        String correlationId = requestContext.getHeaderString(CORRELATION_HEADER);

        if (correlationId == null || correlationId.isBlank()) {
            // Ce service est le premier de la chaîne — génère un nouvel ID
            correlationId = UUID.randomUUID().toString();
        }

        // Mise dans le MDC SLF4J — tous les logs de cette requête incluront le correlation ID
        MDC.put(MDC_KEY, correlationId);
        // Stockage dans le contexte de la requête pour la réponse
        requestContext.setProperty(CORRELATION_HEADER, correlationId);
    }

    // Réponse sortante — retransmet le correlation ID dans la réponse
    @Override
    public void filter(ContainerRequestContext requestContext,
                       ContainerResponseContext responseContext) throws IOException {
        String correlationId = (String) requestContext.getProperty(CORRELATION_HEADER);
        if (correlationId != null) {
            responseContext.getHeaders().add(CORRELATION_HEADER, correlationId);
        }
        // Nettoyage du MDC — évite les fuites de mémoire dans les pools de threads
        MDC.remove(MDC_KEY);
    }
}


// Filtre JAX-RS côté client (MicroProfile Rest Client) — propage le correlation ID
@Provider
public class CorrelationIdClientFilter implements jakarta.ws.rs.client.ClientRequestFilter {

    @Override
    public void filter(jakarta.ws.rs.client.ClientRequestContext requestContext)
            throws IOException {
        // Récupère le correlation ID du MDC et le propage dans les appels sortants
        String correlationId = MDC.get("correlationId");
        if (correlationId != null) {
            requestContext.getHeaders().add("X-Correlation-ID", correlationId);
        }
    }
}
