Entity — Account

java 
import java.util.Objects;

/**
 * Entity — Account
 * Identité : AccountId unique.
 * Mutabilité : le solde et le statut évoluent dans le temps.
 * equals/hashCode : basés sur l'identifiant uniquement.
 */
public class Account {

    private final AccountId id;
    private Money           balance;
    private AccountStatus   status;

    public Account(AccountId id, Money initialBalance) {
        this.id      = Objects.requireNonNull(id, "id must not be null");
        this.balance = Objects.requireNonNull(initialBalance, "balance must not be null");
        this.status  = AccountStatus.ACTIVE;
        validate();
    }

    // --- Comportements métier ---

    public void creditAccount(Money amount) {
        Objects.requireNonNull(amount, "amount must not be null");
        if (amount.isNegativeOrZero()) {
            throw new IllegalArgumentException("Credit amount must be positive");
        }
        this.balance = this.balance.add(amount);
    }

    public void debitAccount(Money amount) {
        Objects.requireNonNull(amount, "amount must not be null");
        if (amount.isNegativeOrZero()) {
            throw new IllegalArgumentException("Debit amount must be positive");
        }
        if (this.balance.isLessThan(amount)) {
            throw new InsufficientFundsException(
                "Insufficient funds: balance=" + balance + ", requested=" + amount);
        }
        if (this.status != AccountStatus.ACTIVE) {
            throw new AccountInactiveException("Account " + id + " is not active");
        }
        this.balance = this.balance.subtract(amount);
    }

    public void close() {
        if (this.status == AccountStatus.CLOSED) {
            throw new IllegalStateException("Account already closed");
        }
        this.status = AccountStatus.CLOSED;
    }

    // --- Validation des invariants ---
    private void validate() {
        if (balance.isNegative()) {
            throw new IllegalArgumentException("Initial balance cannot be negative");
        }
    }

    // --- Accesseurs (lecture seule) ---
    public AccountId    getId()      { return id; }
    public Money        getBalance() { return balance; }
    public AccountStatus getStatus() { return status; }

    // --- equals / hashCode basés sur l'identifiant ---
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Account other)) return false;
        return Objects.equals(id, other.id);
    }

    @Override
    public int hashCode() {
        return Objects.hash(id);
    }

    @Override
    public String toString() {
        return "Account{id=" + id + ", balance=" + balance + ", status=" + status + "}";
    }
}

Value Object — Money

java 
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.Currency;
import java.util.Objects;

/**
 * Value Object — Money
 * Immuabilité : tous les champs final, pas de setters.
 * Égalité : basée sur amount ET currency.
 * Opérations : retournent de nouveaux objets Money.
 */
public final class Money {

    private final BigDecimal amount;
    private final Currency   currency;

    private Money(BigDecimal amount, Currency currency) {
        Objects.requireNonNull(amount,   "amount must not be null");
        Objects.requireNonNull(currency, "currency must not be null");
        if (amount.scale() > 2) {
            throw new IllegalArgumentException("Money scale cannot exceed 2 decimal places");
        }
        this.amount   = amount.setScale(2, RoundingMode.HALF_UP);
        this.currency = currency;
    }

    // --- Factory method ---
    public static Money of(BigDecimal amount, Currency currency) {
        return new Money(amount, currency);
    }

    public static Money of(String amount, String currencyCode) {
        return new Money(new BigDecimal(amount), Currency.getInstance(currencyCode));
    }

    public static Money zero(Currency currency) {
        return new Money(BigDecimal.ZERO, currency);
    }

    // --- Opérations arithmétiques (retournent un nouveau Money) ---
    public Money add(Money other) {
        assertSameCurrency(other);
        return new Money(this.amount.add(other.amount), this.currency);
    }

    public Money subtract(Money other) {
        assertSameCurrency(other);
        return new Money(this.amount.subtract(other.amount), this.currency);
    }

    public Money multiply(BigDecimal factor) {
        return new Money(this.amount.multiply(factor), this.currency);
    }

    // --- Comparaisons ---
    public boolean isNegative()         { return amount.compareTo(BigDecimal.ZERO) < 0; }
    public boolean isNegativeOrZero()   { return amount.compareTo(BigDecimal.ZERO) <= 0; }
    public boolean isPositive()         { return amount.compareTo(BigDecimal.ZERO) > 0; }
    public boolean isLessThan(Money o)  { assertSameCurrency(o); return amount.compareTo(o.amount) < 0; }

    // --- Accesseurs ---
    public BigDecimal getAmount()   { return amount; }
    public Currency   getCurrency() { return currency; }

    // --- Validation devise ---
    private void assertSameCurrency(Money other) {
        if (!this.currency.equals(other.currency)) {
            throw new CurrencyMismatchException(
                "Cannot operate on different currencies: " + currency + " vs " + other.currency);
        }
    }

    // --- equals / hashCode par valeur ---
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Money other)) return false;
        return Objects.equals(amount, other.amount)
            && Objects.equals(currency, other.currency);
    }

    @Override
    public int hashCode() {
        return Objects.hash(amount, currency);
    }

    @Override
    public String toString() {
        return amount.toPlainString() + " " + currency.getCurrencyCode();
    }
}

Value Object — AccountNumber

java 
import java.util.Objects;
import java.util.UUID;
import java.util.regex.Pattern;

/**
 * Value Object — AccountNumber
 * Format : "ACC-" suivi de 8 caractères alphanumériques en majuscules.
 * Immuabilité : final field, pas de setter.
 * Égalité : basée sur la valeur du numéro.
 */
public final class AccountNumber {

    private static final Pattern FORMAT =
        Pattern.compile("^ACC-[A-Z0-9]{8}$");

    private final String value;

    private AccountNumber(String value) {
        validate(value);
        this.value = value;
    }

    // --- Factory methods ---
    public static AccountNumber of(String value) {
        return new AccountNumber(value);
    }

    /** Génère un nouveau numéro de compte unique. */
    public static AccountNumber generate() {
        String uuid = UUID.randomUUID()
            .toString()
            .replace("-", "")
            .substring(0, 8)
            .toUpperCase();
        return new AccountNumber("ACC-" + uuid);
    }

    // --- Validation ---
    private static void validate(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException("AccountNumber must not be blank");
        }
        if (!FORMAT.matcher(value).matches()) {
            throw new IllegalArgumentException(
                "AccountNumber '" + value + "' does not match format ACC-XXXXXXXX");
        }
    }

    // --- Accesseur ---
    public String getValue() { return value; }

    // --- equals / hashCode par valeur ---
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof AccountNumber other)) return false;
        return Objects.equals(value, other.value);
    }

    @Override
    public int hashCode() {
        return Objects.hash(value);
    }

    @Override
    public String toString() {
        return value;
    }
}

Complete Aggregate — Account

java 
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Objects;

/**
 * Aggregate Root — Account
 * Contrôle l'accès à ses Transaction internes.
 * Garantit les invariants : solde, limite, statut.
 * Publie des Domain Events lors des changements d'état.
 */
public class Account {

    private static final Money MAX_TRANSACTION =
        Money.of("10000.00", "EUR");

    private final AccountId         id;
    private final AccountNumber     accountNumber;
    private final ClientId          clientId;      // référence par ID — pas d'objet Client
    private Money                   balance;
    private AccountStatus           status;
    private final List<Transaction> transactions = new ArrayList<>();
    private final List<DomainEvent> domainEvents = new ArrayList<>();

    public Account(AccountId id, AccountNumber accountNumber,
                   ClientId clientId, Money initialBalance) {
        this.id            = Objects.requireNonNull(id);
        this.accountNumber = Objects.requireNonNull(accountNumber);
        this.clientId      = Objects.requireNonNull(clientId);
        this.balance       = Objects.requireNonNull(initialBalance);
        this.status        = AccountStatus.ACTIVE;
        raiseEvent(new AccountOpenedEvent(id, accountNumber, clientId, initialBalance));
    }

    // --- Comportements métier avec invariants ---

    public void addTransaction(Money amount, TransactionType type, String description) {
        // Invariant 1 : le compte doit être actif
        if (status != AccountStatus.ACTIVE) {
            throw new AccountInactiveException("Account " + id + " is not active");
        }
        // Invariant 2 : limite par transaction
        if (amount.isGreaterThan(MAX_TRANSACTION)) {
            throw new TransactionLimitExceededException(
                "Transaction exceeds limit: " + amount + " > " + MAX_TRANSACTION);
        }
        Transaction tx = new Transaction(amount, type, description);
        transactions.add(tx);

        if (type == TransactionType.DEBIT) {
            if (balance.isLessThan(amount)) {
                transactions.remove(tx);
                throw new InsufficientFundsException(
                    "Insufficient funds: " + balance + " < " + amount);
            }
            balance = balance.subtract(amount);
            raiseEvent(new AccountDebitedEvent(id, amount, balance));
        } else {
            balance = balance.add(amount);
            raiseEvent(new AccountCreditedEvent(id, amount, balance));
        }
    }

    // --- Gestion des Domain Events ---

    private void raiseEvent(DomainEvent event) {
        domainEvents.add(event);
    }

    /** Retourne les événements non encore publiés (pour le Repository). */
    public List<DomainEvent> getDomainEvents() {
        return Collections.unmodifiableList(domainEvents);
    }

    /** Appelé par le Repository après publication des événements. */
    public void clearDomainEvents() {
        domainEvents.clear();
    }

    // --- Accesseurs (vues immuables des collections internes) ---
    public AccountId           getId()            { return id; }
    public AccountNumber       getAccountNumber() { return accountNumber; }
    public ClientId            getClientId()      { return clientId; }
    public Money               getBalance()       { return balance; }
    public AccountStatus       getStatus()        { return status; }
    public List<Transaction>  getTransactions()  { return Collections.unmodifiableList(transactions); }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Account other)) return false;
        return Objects.equals(id, other.id);
    }

    @Override
    public int hashCode() {
        return Objects.hash(id);
    }
}

Repository — Domain interface + JPA Implementation

java 
import java.util.List;
import java.util.Optional;

// --- Interface dans le domaine ---
// Parle le langage du domaine, sans mention de JPA ou SQL.
// package: com.esipe.bank.domain.repository
public interface AccountRepository {

    Optional<Account> findById(AccountId accountId);

    List<Account> findByClient(ClientId clientId);

    List<Account> findActiveAccounts();

    Account save(Account account);

    void delete(AccountId accountId);
}


// --- Implémentation dans l'infrastructure ---
// package: com.esipe.bank.infrastructure.jpa
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import jakarta.persistence.TypedQuery;
import jakarta.transaction.Transactional;

@ApplicationScoped
public class JpaAccountRepository implements AccountRepository {

    @PersistenceContext
    private EntityManager em;

    @Override
    public Optional<Account> findById(AccountId accountId) {
        AccountEntity entity = em.find(AccountEntity.class, accountId.getValue());
        return Optional.ofNullable(entity)
                       .map(AccountMapper::toDomain);
    }

    @Override
    public List<Account> findByClient(ClientId clientId) {
        TypedQuery<AccountEntity> query = em.createNamedQuery(
            "AccountEntity.findByClientId", AccountEntity.class);
        query.setParameter("clientId", clientId.getValue());
        return query.getResultList()
                    .stream()
                    .map(AccountMapper::toDomain)
                    .toList();
    }

    @Override
    public List<Account> findActiveAccounts() {
        return em.createNamedQuery("AccountEntity.findActive", AccountEntity.class)
                 .getResultList()
                 .stream()
                 .map(AccountMapper::toDomain)
                 .toList();
    }

    @Override
    @Transactional
    public Account save(Account account) {
        AccountEntity entity = AccountMapper.toEntity(account);
        AccountEntity saved  = em.merge(entity);
        return AccountMapper.toDomain(saved);
    }

    @Override
    @Transactional
    public void delete(AccountId accountId) {
        AccountEntity entity = em.find(AccountEntity.class, accountId.getValue());
        if (entity != null) {
            em.remove(entity);
        }
    }
}

Domain Service — TransferService

java 
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.transaction.Transactional;

/**
 * Domain Service — MoneyTransferService
 * Opération métier impliquant deux aggregates Account.
 * Ni Account A ni Account B ne peut "posséder" cette logique seul.
 * Stateless : aucun champ d'état entre les appels.
 */
@ApplicationScoped
public class MoneyTransferService {

    @Inject
    private AccountRepository accountRepository;

    @Inject
    private DomainEventPublisher eventPublisher;

    /**
     * Effectue un transfert de fonds entre deux comptes.
     * Garantit que les deux comptes existent et que les invariants sont respectés.
     */
    @Transactional
    public void transfer(AccountId fromId, AccountId toId, Money amount) {
        // Charger les deux aggregates
        Account source = accountRepository.findById(fromId)
            .orElseThrow(() -> new AccountNotFoundException(
                "Source account not found: " + fromId));

        Account target = accountRepository.findById(toId)
            .orElseThrow(() -> new AccountNotFoundException(
                "Target account not found: " + toId));

        // Déléguer aux aggregates — ils vérifient leurs propres invariants
        source.addTransaction(amount, TransactionType.DEBIT,
            "Transfer to " + toId);
        target.addTransaction(amount, TransactionType.CREDIT,
            "Transfer from " + fromId);

        // Persister les deux aggregates modifiés
        accountRepository.save(source);
        accountRepository.save(target);

        // Publier le Domain Event de complétion du transfert
        TransferCompletedEvent event = new TransferCompletedEvent(
            fromId, toId, amount, java.time.Instant.now());
        eventPublisher.publish(event);
    }
}

public interface DomainEventPublisher {

import java.time.Instant;

// Nommé au passé : quelque chose qui s'est produit dans le domaine.
public record TransferCompletedEvent(
    AccountId accountFrom,
    AccountId accountTo,
    Money     amount,
    Instant   occurredOn
) implements DomainEvent {

    // Compact constructor — validation
    public TransferCompletedEvent {
        if (accountFrom == null) throw new IllegalArgumentException("accountFrom required");
        if (accountTo   == null) throw new IllegalArgumentException("accountTo required");
        if (amount      == null) throw new IllegalArgumentException("amount required");
        if (occurredOn  == null) occurredOn = Instant.now();
    }
}


// --- Marqueur commun pour tous les Domain Events ---
public interface DomainEvent {
    Instant occurredOn();
}

Java 17 Records in DDD

Java Records are ideal for Value Objects: immutable and concise. They automatically generate constructor, accessors, equals/hashCode and toString — based on all components.

Note: Java Records support as @Embeddable has been added experimentally in Hibernate 6.x and EclipseLink 4.x. Check your JPA provider documentation for current support status.

java 
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.Currency;
import java.util.Objects;

/**
 * Value Object Money — implémenté en Java Record (Java 17).
 * Le record génère automatiquement : constructeur, accesseurs,
 * equals/hashCode et toString — basés sur tous les composants.
 * Idéal pour les Value Objects : immuable et concis.
 */
public record Money(BigDecimal amount, Currency currency) {

    // Compact constructor — exécuté à la place du constructeur généré
    public Money {
        Objects.requireNonNull(amount,   "amount must not be null");
        Objects.requireNonNull(currency, "currency must not be null");
        if (amount.scale() > 2) {
            throw new IllegalArgumentException(
                "Money scale cannot exceed 2 decimal places");
        }
        // Normaliser l'échelle — réassignation autorisée dans un compact constructor
        amount = amount.setScale(2, RoundingMode.HALF_UP);
    }

    // --- Factory methods statiques ---
    public static Money of(BigDecimal amount, Currency currency) {
        return new Money(amount, currency);
    }

    public static Money of(String amount, String currencyCode) {
        return new Money(new BigDecimal(amount), Currency.getInstance(currencyCode));
    }

    public static Money euros(String amount) {
        return Money.of(amount, "EUR");
    }

    public static Money zero(Currency currency) {
        return new Money(BigDecimal.ZERO, currency);
    }

    // --- Opérations métier (retournent un nouveau record) ---
    public Money add(Money other) {
        assertSameCurrency(other);
        return new Money(this.amount.add(other.amount), this.currency);
    }

    public Money subtract(Money other) {
        assertSameCurrency(other);
        return new Money(this.amount.subtract(other.amount), this.currency);
    }

    public Money multiply(BigDecimal factor) {
        return new Money(this.amount.multiply(factor), this.currency);
    }

    // --- Prédicats métier ---
    public boolean isNegative()       { return amount.signum() < 0; }
    public boolean isPositive()       { return amount.signum() > 0; }
    public boolean isZero()           { return amount.signum() == 0; }
    public boolean isLessThan(Money o){ assertSameCurrency(o); return amount.compareTo(o.amount) < 0; }
    public boolean isGreaterThan(Money o){ assertSameCurrency(o); return amount.compareTo(o.amount) > 0; }

    // --- Validation ---
    private void assertSameCurrency(Money other) {
        if (!this.currency.equals(other.currency)) {
            throw new CurrencyMismatchException(this.currency + " vs " + other.currency);
        }
    }

    // toString personnalisé (remplace celui généré par le record)
    @Override
    public String toString() {
        return amount.toPlainString() + " " + currency.getCurrencyCode();
    }
}
